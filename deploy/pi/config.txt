[all]
# Full KMS graphics (the deck draws with GLES 3 straight to the display),
# without HDMI audio; and no onboard audio: the DAC hat is the only output.
# install.sh comments out the stock lines these replace.
dtoverlay=vc4-kms-v3d,noaudio
dtparam=audio=off

# The PCM5122 DAC hat, clocked by the Pi (-std): right on any PCM5122 board.
# Plain hifiberry-dacplus trusts the board's own clock chips, and on a board
# whose chips aren't a DAC+ Pro's everything plays at the wrong speed (ours:
# twice as fast). kdj-dac times the hat at boot and switches between the two
# by itself if this one doesn't keep time.
dtoverlay=hifiberry-dacplus-std

# The display. HDMI screens need nothing here (set the mode in cmdline.txt,
# see install.sh). For a DSI panel, uncomment the line for yours, e.g.:
#dtoverlay=vc4-kms-dsi-7inch
#dtoverlay=vc4-kms-dsi-waveshare-panel,5_0_inch

# No Bluetooth: nothing to pair mid-set, and the UART it used is free.
dtoverlay=disable-bt

# Quiet, quick boot: no rainbow splash (boot messages go to tty3, see
# cmdline.txt).
disable_splash=1
boot_delay=0

#kernel=kernel8-rt.img
