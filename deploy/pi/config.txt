[all]
# Full KMS graphics (the deck draws with GLES 3 straight to the display),
# without HDMI audio; and no onboard audio: the DAC hat is the only output.
# setup.sh comments out the stock lines these replace.
dtoverlay=vc4-kms-v3d,noaudio
dtparam=audio=off

# The PCM5122 DAC hat. Most PCM5122 boards work as a HiFiBerry DAC+;
# boards without its clock chip want hifiberry-dacplus-std, and IQaudio
# boards iqaudio-dacplus. Check your board's docs.
dtoverlay=hifiberry-dacplus

# The display. HDMI screens need nothing here (set the mode in cmdline.txt,
# see setup.sh). For a DSI panel, uncomment the line for yours, e.g.:
#dtoverlay=vc4-kms-dsi-7inch
#dtoverlay=vc4-kms-dsi-waveshare-panel,5_0_inch

# Quiet, quick boot.
disable_splash=1
boot_delay=0
